void MainComponent::updateChannelButtonStates()
{
    for (int i = 0; i < NUM_CHANNELS; ++i)
    {
        bool active = (i == activeChannel);
        channelButtons[i].setColour (juce::TextButton::buttonColourId,
            active ? juce::Colour (0xff4040a0) : juce::Colour (0xff2a2a4a));
        channelButtons[i].setColour (juce::TextButton::textColourOffId,
            active ? juce::Colours::white : juce::Colours::lightgrey);
    }
    repaint();
}

--- REPLACE ---

void MainComponent::updateChannelButtonStates()
{
    for (int i = 0; i < NUM_CHANNELS; ++i)
    {
        const bool isActive = (i == activeChannel);
        const bool isAbA    = (i == abChannelA);
        const bool isAbB    = (i == abChannelB);

        // Active channel: bright blue.
        // A bookmark but not active: dim amber tint.
        // B bookmark but not active: dim teal tint.
        // Default: dark background.
        juce::Colour bg;
        if (isActive)      bg = juce::Colour (0xff4040a0);          // active
        else if (isAbA)    bg = juce::Colour (0xff7a5a10);          // A bookmarked
        else if (isAbB)    bg = juce::Colour (0xff10607a);          // B bookmarked
        else               bg = juce::Colour (0xff2a2a4a);          // idle

        channelButtons[i].setColour (juce::TextButton::buttonColourId, bg);
        channelButtons[i].setColour (juce::TextButton::textColourOffId,
            isActive ? juce::Colours::white : juce::Colours::lightgrey);
    }

    // Update the toggle button label to show which side is currently live
    // and which side pressing it will jump to.
    abToggleButton.setButtonText (abIsShowingA ? "A/B \xe2\x86\x92 B"   // → B
                                               : "A/B \xe2\x86\x92 A"); // → A

    // Highlight the toggle button when actively showing B (non-default side).
    abToggleButton.setColour (juce::TextButton::buttonColourId,
        abIsShowingA ? juce::Colour (0xff2a2a4a) : juce::Colour (0xff204060));

    repaint();
}
